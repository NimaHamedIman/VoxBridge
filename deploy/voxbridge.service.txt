[Unit]
Description=VoxBridge AI Assistant API
After=network.target

[Service]
Type=simple
# Dedicated non-root user (least privilege)
User=voxbridge
Group=voxbridge
WorkingDirectory=/opt/VoxBridge
# Whisper/HuggingFace cache must live somewhere the service user can write.
Environment=XDG_CACHE_HOME=/opt/VoxBridge/.cache
# PATH must include the system directories, otherwise Whisper
# cannot find ffmpeg (systemd does not inherit the login PATH).
Environment=PATH=/opt/VoxBridge/venv/bin:/usr/local/bin:/usr/bin:/bin
ExecStart=/opt/VoxBridge/venv/bin/uvicorn src.api:app --host 127.0.0.1 --port 8000
Restart=always
RestartSec=3

[Install]
WantedBy=multi-user.target
